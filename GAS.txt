function doPost(e) {
  // Tu función parsePostData(e) es correcta, la mantenemos.
  function parsePostData(e) {
     // ... (tu código de parsePostData sin cambios) ...
  if (!e) return {};
 
  if (e.postData && e.postData.type === 'application/json') {
   try { 
     return JSON.parse(e.postData.contents); 
   } catch (_) { 
     return {}; 
   }
  }
  
  if (e.postData) {
   try { 
     return JSON.parse(e.postData.contents); 
   }
   catch (err) {
     const obj = {};
     const pairs = e.postData.contents.split('&');
     for (let i = 0; i < pairs.length; ++i) {
       const parts = pairs[i].split('=');
       const key = decodeURIComponent(parts[0]);
       const value = decodeURIComponent((parts[1] || '').replace(/\+/g, ' '));
       obj[key] = value;
     }
     return obj;
   }
  }
  
  if (e.parameter && Object.keys(e.parameter).length > 0) return e.parameter;
  return {};
  }


  try {
    const data = parsePostData(e);
    Logger.log('Datos recibidos en doPost: ' + JSON.stringify(data));
    if (data.action && data.action !== 'saveFormData') {
      return handleCategoriasAction_(data.action, data, 'POST');
    }

    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Hoja 1');
    if (!sheet) throw new Error('Hoja no encontrada');
    const tz = SpreadsheetApp.getActive().getSpreadsheetTimeZone() || 'Europe/Madrid';
    
    // Tu lista expectedHeaders es correcta, la mantenemos.
    const expectedHeaders = [
       'your-name','your-email','tel-686','checkbox-374','cualidades-otros',
       'radio-188','ubicacion-terreno','number-419','date-33','number-420',
       'number-421','number-422','number-423','coherencia-dormitorios',
       'coherencia-presupuesto','Fecha','Respondido','Notas','FechaSeguimiento',
       'Prioridad','Estado','Llamado','ID','fecha-llamada','como-conocido',
       'como-conocido-otros','descripcion-vivienda','distribucion-dia','garaje',
       'piscina','estancia-adicional','superficie-parcela','edificabilidad',
       'ocupacion','referencia-catastral','presupuesto-deseado','viabilidad',
       'informacion-adicional','fecha-mail','info-enviada','fecha-reunion',
       'imprescindible','NoContestados','procedencia-contacto','vivienda-interesada','documentacion','eliminado','MotivoSeguimiento','FechaNotificacion','HoraNotificacion','estudio-viabilidad','categoria-id'
     ];

    // Lógica de forzar cabeceras (mantenemos)
    let currentHeaders = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    if (currentHeaders.length < expectedHeaders.length || 
        currentHeaders.some((h, i) => String(h) !== expectedHeaders[i] && i < expectedHeaders.length)) {
      sheet.getRange(1, 1, 1, expectedHeaders.length).setValues([expectedHeaders]);
      currentHeaders = expectedHeaders;
    }
    
    // Generar o usar ID
    const isWordpress = !data.action && !data.source && data['your-name'];
    let id = data.id || data.ID || `contacto_${Date.now()}_${Math.floor(Math.random() * 1000)}`;
    Logger.log('ID usado para fila: ' + id);

    // --- Lógica de Actualizar vs. Añadir (OPTIMIZADA) ---
    
    // Buscar fila existente
    const rowNum = findRowById_(sheet, id, currentHeaders);
    let updated = (rowNum !== -1);
    let existingData = {};

    if (updated) {
      // Si actualizamos, leemos SOLO los datos existentes de esa fila
      const existingRowValues = sheet.getRange(rowNum, 1, 1, currentHeaders.length).getValues()[0];
      existingData = convertRowToObject_(currentHeaders, existingRowValues);
    }
    
    // Fusionar datos: los nuevos (data) sobreescriben a los viejos (existingData)
    const finalData = { ...existingData, ...data };
    if (!finalData['categoria-id']) {
      finalData['categoria-id'] = resolveCategoriaIdDesdeDatos_(finalData);
    }
    
    // Mapear la fila completa según headers
    let rowData = expectedHeaders.map(h => {
      if (h === 'ID') return id;
      // Tu lógica de valores por defecto (es correcta)
      if (h === 'Respondido') return finalData[h] || 'No';
      if (h === 'Estado') return finalData[h] || 'Sin Llamar';
    
      if (h === 'Llamado') return finalData[h] || 'No';
      // ... (resto de tu lógica de valores por defecto) ...
      if (h === 'Fecha') {
        if (isWordpress) {
          return Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd');
        }
        let v = finalData[h] || finalData['date-33'];
        return v || '';
      }
      if (h === 'checkbox-374') {
         if (Array.isArray(finalData[h])) return finalData[h].join(',');
         if (typeof finalData[h] === 'object') return Object.values(finalData[h]).join(',');
      }
      return finalData[h] || ''; // Usar el dato fusionado
    });

    // Escribir en la hoja
    if (updated) {
      sheet.getRange(rowNum, 1, 1, expectedHeaders.length).setValues([rowData]);
      Logger.log('Fila actualizada en posición: ' + rowNum);
    } else {
      sheet.appendRow(rowData);
      Logger.log('Fila añadida nueva');
    }

    // --- CORRECCIÓN BUG: Devolver la fila actualizada/creada ---
    const dataObj = convertRowToObject_(expectedHeaders, rowData);
    
    return ContentService.createTextOutput(JSON.stringify({ 
      status: 'success', 
      message: updated ? 'Fila actualizada' : 'Fila añadida', 
      id: id,
      data: dataObj // Devolver los datos completos
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    Logger.log('Error en doPost: ' + error.message + '\nStack: ' + error.stack);
    return createResponse({ status: 'error', message: error.message });
  }
}

function handleCategoriasAction_(action, payload, method) {
  try {
    const cfg = getCategoriasConfig_();
    const categorias = cfg.categorias || [];
    const actionName = String(action || '').trim();

    if (actionName === 'getConfig') {
      return createResponse({ status: 'success', config: cfg });
    }

    if (actionName === 'saveConfig') {
      const parsedConfig = parseConfigPayload_(payload.config);
      const normalizada = normalizeCategoriasConfig_(parsedConfig || cfg);
      saveCategoriasConfig_(normalizada);
      return createResponse({ status: 'success', config: normalizada });
    }

    if (actionName === 'addCategoria') {
      const nombre = String(payload.nombre || '').trim();
      if (!nombre) return createResponse({ status: 'error', message: 'Nombre obligatorio' });
      const id = ensureCategoriaIdUnico_(slugify_(nombre), categorias);
      const nueva = {
        id: id,
        nombre: nombre,
        activa: true,
        matcher: { by: 'manual', aliases: [nombre], keywords: [], value: '' },
        formulario: { tipo: 'standard', campos: [], detalle: { tipo: 'standard', campos: [] } },
        detalle: { tipo: 'standard', campos: [] },
        email: { asunto: '', cuerpo: '', recordatorio: '' },
        exportacion: { tipo: 'normal', prefijoArchivo: nombre.replace(/\s+/g, '_') },
        wordpress: { flujo: 'default' }
      };
      upsertCategoriaInSheet_(nueva);
      const next = getCategoriasConfig_();
      return createResponse({ status: 'success', categoria: nueva, config: next });
    }

    if (actionName === 'updateCategoria') {
      const id = String(payload.id || '').trim();
      if (!id) return createResponse({ status: 'error', message: 'ID obligatorio' });
      const idx = categorias.findIndex(c => String(c.id) === id);
      if (idx === -1) return createResponse({ status: 'error', message: 'Categoría no encontrada' });
      const categoriaPatch = parseCategoriaPatch_(payload);
      const baseProtegida = isCategoriaBaseProtegida_(id);
      if (baseProtegida && categoriaPatch.id && categoriaPatch.id !== id) {
        return createResponse({ status: 'error', message: 'No se puede cambiar el ID de una categoría base' });
      }
      if (baseProtegida && categoriaPatch.nombre && categoriaPatch.nombre !== categorias[idx].nombre) {
        return createResponse({ status: 'error', message: 'No se puede renombrar una categoría base' });
      }
      const camposActuales = Array.isArray(categorias[idx].formulario?.campos) ? categorias[idx].formulario.campos : [];
      const camposNuevos = Array.isArray(categoriaPatch.formulario?.campos) ? categoriaPatch.formulario.campos : null;
      if (camposNuevos) {
        const eliminados = camposActuales.filter(c => camposNuevos.indexOf(c) === -1);
        const campoConDatos = eliminados.find(campo => categoriaFieldHasData_(campo));
        if (campoConDatos) {
          return createResponse({
            status: 'error',
            message: `No se puede eliminar el campo "${campoConDatos}" porque ya tiene datos en fichas existentes`
          });
        }
      }
      categorias[idx] = { ...categorias[idx], ...categoriaPatch, id: id };
      upsertCategoriaInSheet_(categorias[idx]);
      const next = getCategoriasConfig_();
      return createResponse({ status: 'success', categoria: categorias[idx], config: next });
    }

    if (actionName === 'deleteCategoria') {
      const id = String(payload.id || '').trim();
      if (!id) return createResponse({ status: 'error', message: 'ID obligatorio' });
      if (isCategoriaBaseProtegida_(id)) {
        return createResponse({ status: 'error', message: 'No se puede eliminar una categoría base protegida' });
      }
      deleteCategoriaFromSheet_(id);
      const next = getCategoriasConfig_();
      return createResponse({ status: 'success', config: next });
    }

    if (actionName === 'toggleCategoria') {
      const id = String(payload.id || '').trim();
      const idx = categorias.findIndex(c => String(c.id) === id);
      if (idx === -1) return createResponse({ status: 'error', message: 'Categoría no encontrada' });
      categorias[idx].activa = !parseBoolean_(categorias[idx].activa, true);
      upsertCategoriaInSheet_(categorias[idx]);
      const next = getCategoriasConfig_();
      return createResponse({ status: 'success', categoria: categorias[idx], config: next });
    }

    if (actionName === 'saveCampaign') {
      const campaign = parseCampaignPayload_(payload.campaign || payload);
      return saveCampaign_(campaign);
    }

    if (actionName === 'updateCampaign') {
      const campaign = parseCampaignPayload_(payload.campaign || payload);
      return updateCampaign_(campaign);
    }

    if (actionName === 'deleteCampaign') {
      const id = String(payload.id || '').trim();
      return deleteCampaign_(id);
    }

    if (actionName === 'getCampaigns') {
      return createResponse({ status: 'success', data: getCampaigns_() });
    }

    return createResponse({ status: 'error', message: 'Acción no soportada' });
  } catch (error) {
    Logger.log(`Error en handleCategoriasAction_: ${error.message}`);
    return createResponse({ status: 'error', message: error.message });
  }
}

function getCategoriasConfig_() {
  const sheet = getCategoriasSheet_();
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) {
    const defaultCfg = normalizeCategoriasConfig_(buildDefaultCategoriasConfig_());
    saveCategoriasConfig_(defaultCfg);
    return defaultCfg;
  }
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  const rows = sheet.getRange(2, 1, lastRow - 1, headers.length).getValues();
  const categorias = rows
    .map(row => rowToCategoria_(headers, row))
    .filter(c => c && c.id);
  return normalizeCategoriasConfig_({ categorias: categorias });
}

function rowToCategoria_(headers, row) {
  const raw = {};
  headers.forEach((h, idx) => raw[h] = row[idx]);
  if (!raw.id) return null;
  const parseObj = (val, fallback) => {
    if (!val) return fallback;
    if (typeof val === 'object') return val;
    try {
      return JSON.parse(String(val));
    } catch (_) {
      return fallback;
    }
  };
  return {
    id: String(raw.id),
    nombre: String(raw.nombre || raw.id),
    activa: parseBoolean_(raw.activa, true),
    matcher: parseObj(raw.matcher, { by: 'manual', aliases: [], keywords: [], value: '' }),
    formulario: parseObj(raw.formulario, { tipo: 'standard', campos: [] }),
    detalle: parseObj(raw.detalle, parseObj(raw.formulario, { detalle: { tipo: 'standard', campos: [] } }).detalle || { tipo: 'standard', campos: [] }),
    email: parseObj(raw.email, { asunto: '', cuerpo: '', recordatorio: '' }),
    exportacion: parseObj(raw.exportacion, { tipo: 'normal', prefijoArchivo: '' }),
    wordpress: parseObj(raw.wordpress, { flujo: 'default' })
  };
}

function categoriaToRow_(headers, categoria) {
  const safe = {
    id: categoria.id,
    nombre: categoria.nombre || categoria.id,
    activa: parseBoolean_(categoria.activa, true),
    matcher: JSON.stringify(categoria.matcher || {}),
    formulario: JSON.stringify(categoria.formulario || {}),
    email: JSON.stringify(categoria.email || {}),
    exportacion: JSON.stringify(categoria.exportacion || {}),
    wordpress: JSON.stringify(categoria.wordpress || {}),
    updatedAt: Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'Europe/Madrid', 'yyyy-MM-dd HH:mm:ss'),
    detalle: JSON.stringify(categoria.detalle || categoria.formulario?.detalle || { tipo: 'standard', campos: [] })
  };
  return headers.map(h => safe[h] !== undefined ? safe[h] : '');
}

function ensureCategoriasHeaders_(sheet) {
  const expected = ['id', 'nombre', 'activa', 'matcher', 'formulario', 'email', 'exportacion', 'wordpress', 'updatedAt', 'detalle'];
  const lastCol = Math.max(sheet.getLastColumn(), expected.length);
  const current = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
  const needsReset = expected.some((h, i) => String(current[i] || '') !== h);
  if (needsReset) {
    sheet.getRange(1, 1, 1, expected.length).setValues([expected]);
  }
  return expected;
}

function findCategoriaRowById_(sheet, id, headers) {
  const idIndex = headers.indexOf('id');
  if (idIndex === -1 || sheet.getLastRow() < 2) return -1;
  const range = sheet.getRange(2, idIndex + 1, sheet.getLastRow() - 1, 1);
  const found = range.createTextFinder(String(id)).matchEntireCell(true).findNext();
  return found ? found.getRow() : -1;
}

function upsertCategoriaInSheet_(categoria) {
  const sheet = getCategoriasSheet_();

  const row = categoriaToRow_(headers, categoria);

  if (rowNum === -1) {

  }
}

function deleteCategoriaFromSheet_(id) {

function parseCategoriaPatch_(payload) {
  let fromJson = {};
  if (payload.categoria) {
    try {
      fromJson = typeof payload.categoria === 'string' ? JSON.parse(payload.categoria) : payload.categoria;
    } catch (_) {}
  }
  const patch = { ...fromJson };
  ['nombre', 'activa', 'matcher', 'formulario', 'detalle', 'email', 'exportacion', 'wordpress'].forEach(k => {
    if (payload[k] !== undefined && payload[k] !== '') {
      patch[k] = payload[k];
    }
  });
  ['matcher', 'formulario', 'detalle', 'email', 'exportacion', 'wordpress'].forEach((key) => {
    if (typeof patch[key] === 'string') {
      try { patch[key] = JSON.parse(patch[key]); } catch (_) {}
    }
  });
  if (typeof patch.activa === 'string') {
    patch.activa = parseBoolean_(patch.activa, true);
  }
  if (!patch.detalle && patch.formulario && patch.formulario.detalle) {
    patch.detalle = patch.formulario.detalle;
  }
  if (patch.detalle && (!patch.formulario || !patch.formulario.detalle)) {
    patch.formulario = { ...(patch.formulario || {}), detalle: patch.detalle };
  }
  return patch;
}

function slugify_(text) {
  return String(text || '')
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/[^a-z0-9]+/g, '_')
    .replace(/^_+|_+$/g, '') || 'categoria';
}

function ensureCategoriaIdUnico_(idBase, categorias) {
  const usados = new Set((categorias || []).map(c => String(c.id)));
  if (!usados.has(idBase)) return idBase;
  let i = 2;
  while (usados.has(`${idBase}_${i}`)) i++;
  return `${idBase}_${i}`;
}

function parseBoolean_(value, fallback) {
  if (typeof value === 'boolean') return value;
  if (value === null || value === undefined || value === '') return fallback !== undefined ? !!fallback : true;
  const normalized = normalizeString_(value);
  if (['false', '0', 'no', 'off', 'oculta', 'inactiva'].indexOf(normalized) !== -1) return false;
  if (['true', '1', 'si', 'sí', 'on', 'activa'].indexOf(normalized) !== -1) return true;
  return fallback !== undefined ? !!fallback : true;
}

function formatFechaOut_(v, tz) {
  if (Object.prototype.toString.call(v) === '[object Date]' && !isNaN(v)) {
    return Utilities.formatDate(v, tz, 'yyyy-MM-dd');
  }
  const s = String(v || '');
  const mISO = s.match(/^(\d{4}-\d{2}-\d{2})T/); // ISO con hora -> tomar solo YYYY-MM-DD
  if (mISO) return mISO[1];
  if (/^\d{2}\/\d{2}\/\d{4}$/.test(s)) {         // dd/mm/yyyy -> yyyy-mm-dd
    const [dd, mm, yyyy] = s.split('/');
    return `${yyyy}-${mm}-${dd}`;
  }
  if (/^\d{4}-\d{2}-\d{2}$/.test(s)) return s;   // ya correcto
  return s;
}

/**
 * Encuentra el número de fila de un contacto por su ID usando TextFinder.
 * @param {Sheet} sheet La hoja donde buscar.
 * @param {string} id El ID a encontrar.
 * @param {Array<string>} headers Array de cabeceras (para encontrar idIndex).
 * @returns {number} El número de fila (base 1) o -1 si no se encuentra.
 */
function findRowById_(sheet, id, headers) {
  const idIndex = headers.indexOf('ID');
  if (idIndex === -1) {
    Logger.log('Error: Columna ID no encontrada en findRowById_');
    return -1;
  }
  
  // Buscar solo en la columna de ID (columna idIndex + 1)
  const idColumnRange = sheet.getRange(2, idIndex + 1, sheet.getLastRow() - 1, 1);
  const textFinder = idColumnRange.createTextFinder(String(id)).matchEntireCell(true);
  const found = textFinder.findNext();
  
  if (found) {
    return found.getRow(); // Devuelve el número de fila exacto
  } else {
    return -1;
  }
}

/**
 * Convierte un array de fila (rowData) y un array de cabeceras (headers) en un objeto JSON.
 * @param {Array<string>} headers - Las cabeceras.
 * @param {Array<any>} rowData - Los datos de la fila.
 * @returns {Object} El objeto JSON de la fila.
 */
function convertRowToObject_(headers, rowData) {
  const obj = {};
  headers.forEach((h, j) => {
    obj[h] = (rowData[j] !== undefined) ? rowData[j] : '';
  });
  return obj;
}

function resolveCategoriaIdDesdeDatos_(datos) {
  const categoriaId = String(datos['categoria-id'] || '').trim();
  if (categoriaId) return categoriaId;
  const interes = normalizeString_(datos.interes || datos['interes-select'] || datos['interes-otros']);
  const viviendaInteresada = normalizeString_(datos['vivienda-interesada']);
  const cfg = getCategoriasConfig_();
  const categorias = cfg.categorias || [];

  const byInteres = categorias.find((cat) => {
    const expected = normalizeString_((cat.matcher && cat.matcher.value) || cat.nombre || '');
    return normalizeString_(cat.matcher && cat.matcher.by) === 'interes' && expected && expected === interes;
  });
  if (byInteres) return byInteres.id;

  const byVivienda = categorias.find((cat) => {
    if (normalizeString_(cat.matcher && cat.matcher.by) !== 'vivienda-interesada') return false;
    const keywords = Array.isArray(cat.matcher && cat.matcher.keywords) ? cat.matcher.keywords : [];
    return keywords.some(k => viviendaInteresada.includes(normalizeString_(k)));
  });
  if (byVivienda) return byVivienda.id;

  const byNombre = categorias.find((cat) => {
    const alias = Array.isArray(cat.matcher && cat.matcher.aliases) ? cat.matcher.aliases : [];
    const keys = [cat.nombre].concat(alias).map(normalizeString_).filter(Boolean);
    return keys.some(k => interes === k || interes.includes(k) || viviendaInteresada.includes(k));
  });
  if (byNombre) return byNombre.id;

  const byDefault = categorias.find(cat => normalizeString_(cat.matcher && cat.matcher.by) === 'default');
  return byDefault ? byDefault.id : '';
}

function normalizeString_(value) {
  return String(value || '').trim().toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
}

  const headers = ensureCampaignHeaders_(sheet);
  const id = String((campaign && campaign.id) || '').trim();
  if (!id) return createResponse({ status: 'error', message: 'ID de campaña obligatorio' });
  if (sheet.getLastRow() < 2) return createResponse({ status: 'error', message: 'No hay campañas guardadas' });

  const ids = sheet.getRange(2, 1, sheet.getLastRow() - 1, 1).getValues().map(r => String(r[0]));
  const idx = ids.findIndex(v => v === id);
  if (idx === -1) return createResponse({ status: 'error', message: 'Campaña no encontrada' });

  const rowNum = idx + 2;
  const row = headers.map(h => (campaign[h] !== undefined ? campaign[h] : ''));
  row[0] = id;
  sheet.getRange(rowNum, 1, 1, headers.length).setValues([row]);
  return createResponse({ status: 'success', id: id, campaign: campaign });
}

function deleteCampaign_(id) {
  const campaignId = String(id || '').trim();
  if (!campaignId) return createResponse({ status: 'error', message: 'ID de campaña obligatorio' });
  const sheet = getCampaignSheet_();
  if (sheet.getLastRow() < 2) return createResponse({ status: 'error', message: 'No hay campañas guardadas' });
  const ids = sheet.getRange(2, 1, sheet.getLastRow() - 1, 1).getValues().map(r => String(r[0]));
  const idx = ids.findIndex(v => v === campaignId);
  if (idx === -1) return createResponse({ status: 'error', message: 'Campaña no encontrada' });
  sheet.deleteRow(idx + 2);
  return createResponse({ status: 'success', id: campaignId });
}

function getCampaigns_() {
  const sheet = getCampaignSheet_();
  if (sheet.getLastRow() < 2) return [];
  const headers = ensureCampaignHeaders_(sheet);
  const values = sheet.getRange(2, 1, sheet.getLastRow() - 1, headers.length).getValues();
  return values.map(row => {
    const obj = {};
    headers.forEach((h, i) => obj[h] = row[i]);
    return obj;
  });
}
